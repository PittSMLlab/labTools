0748             %UNTANGLELIKELIHOODS Assign statistic likelihoods to markers.
0749             %
0750             %   markerScores = UNTANGLELIKELIHOODS(L,indicatrix) finds, for
0751             % each frame, marker deviations x >= 0 (in standard deviations)
0752             % that explain the statistic likelihoods: a linear program
0753             % minimizes sum(x) subject to indicatrix'*x >= sqrt(-2*L); among
0754             % solutions with the same active constraints, fmincon then picks
0755             % the one with the largest sum(x.^2), that is, the most unequal
0756             % one. The score is -0.5*x.^2. The tolerances are relaxed for
0757             % speed. Prints the number of frames and the elapsed time.
0758             %
0759             % Inputs:
0760             %   L          - statistics x frames log-likelihoods
0761             %   indicatrix - markers x statistics matrix, 1 where the
0762             %                statistic involves the marker
0763             %
0764             % Outputs:
0765             %   markerScores - markers x frames scores
0766             %
0767             % Toolbox Dependencies: Optimization Toolbox (fmincon, linprog,
0768             %   optimoptions)
0769 
0770             A = -indicatrix';
0771             %Relax tolerances for fast compute
0772             relaxedTol = 1e-1;
0773             options = optimoptions('fmincon','Display','off', ...
0774                 'SpecifyObjectiveGradient',false,'OptimalityTolerance', ...
0775                 relaxedTol,'StepTolerance',relaxedTol, ...
0776                 'ConstraintTolerance',relaxedTol);
0777             options2 = optimoptions('linprog','Display','off');
0778             N = size(indicatrix,1);
0779             LB = zeros(N,1);
0780             f = ones(N,1);
0781             M = size(L,2); %number of frames
0782             markerScores = nan(N,M);
0783             display(['Running for ' num2str(M) ' frames. Expect ' ...
0784                 num2str(M/75,2) ' sec. processing time (75 fps).'])
0785             tic
0786             for fr = 1:M
0787                 %transform likelihood to std away from expectation
0788                 b = -sqrt(-2*L(:,fr));
0789                 %Get linear programming solution:
0790                 %Being used to identify active inequalities only
0791                 [x0,~,~,~,lam] = linprog(f,A,b,[],[],LB,[],options2);
0792                 %f'*x0
0793                 %Sparsify solution: (for each active constraint, selecting
0794                 %the most
0795                 %unequal solution possible, notice this maintains the cost
0796                 %f'*x constant)
0797                 activeConstraints = (lam.ineqlin == 1);
0798                 A1 = -A(activeConstraints,:); %Active inequalities
0799                 %For each marker there is at least 1 active inequality, or
0800                 %one of the bounds was reached
0801                 idx = any(A1 ~= 0);

0803                 %Notice this adds as constraint that active inequalities
0804                 %from linprog are maintained

0807                     A2*x0(idx),LB(idx),[],[],options);

0817             %
0818             %   outMarkers = UNTANGLEOUTLIERS(outStats,in) solves, for each
0819             % frame, min 1'*y s.t. outStats <= in'*y and y >= (in*outStats -

0821             % statistics per marker. This finds the minimum outlier set such

0823             % causing it, and any marker with more outlier statistics than
0824             % there are outlier markers is itself an outlier. The program is
0825             % only solved for a frame when the initial set (markers with
0826             % more than one outlier statistic) violates these constraints or
0827             % has more than one marker.

0829             % Inputs:
0830             %   outStats - statistics x frames logical, true for outlier
0831             %              statistics
0832             %   in       - markers x statistics indicatrix
0833             %
0834             % Outputs:
0835             %   outMarkers - markers x frames logical, true for outlier
0836             %                markers
0837             %
0838             % Toolbox Dependencies: Optimization Toolbox (intlinprog,
0839             %   optimoptions)
0840 
0841             [M,P] = size(in);
0842             uno        = ones(M,1);
0843             cero       = zeros(M,1);
0844             f          = [uno]; %Vector to solve y
0845             lb         = [cero];
0846             ub         = [uno];
0847             opts       = optimoptions('intlinprog','Display','off');
0848             %This marks markers only if they have more than one outlier stat
0849             outMarkers = (in*outStats) > 1;

0851                 b1 = outStats(:,fr);
