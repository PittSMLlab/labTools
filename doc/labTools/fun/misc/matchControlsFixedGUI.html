matchControlsFixedGUI

PURPOSE

MATCHCONTROLSFIXEDGUI Match control participants to a target participant.

SYNOPSIS

function matchControlsFixedGUI(demoFile)

DESCRIPTION 

MATCHCONTROLSFIXEDGUI Match control participants to a target participant.

   Opens a GUI to choose the demographic matching factors (Sex, Age via
 DOB, Height, Weight, Years formal education, Race, Ethnicity), enter a
 target participant, and list the closest candidates in a demographics
 table. Categorical factors filter the candidates; numeric factors are
 ranked by a weighted, tolerance-normalized Euclidean distance. Target
 and candidate height/weight are converted to a canonical unit (see
 canonicalUnits) before distances and tolerances are computed. Rows
 already marked as matched are excluded. The five closest candidates are
 shown with a table of which numeric factors fall within tolerance.
 This tool was created via ChatGPT. Needs a MATLAB release with
 uidatepicker (newer than R2021a).

 Inputs:
   demoFile - (optional) Path to a demographics .xlsx, .xls, or .csv
              file; opens a file dialog when empty (default '')

 Outputs:
   None

 Side effects: opens the GUI figure and, from its buttons, table
   windows; reads demoFile; nothing is written.

 Example:
   matchControlsFixedGUI()                      % choose file with dialog
   matchControlsFixedGUI('Demographics.xlsx')   % pass path

 Toolbox Dependencies: None

 See also READTABLE, UIFIGURE, UITABLE.

0508 %% helper function: normalize categorical values for reliable comparison
0509 % - special handling for Sex: maps common short codes to canonical labels.
0510 % - for other categorical variables we use trimmed lowercase strings.
0511     function out = normalizeCategorical(vals,varKey)
0512         %NORMALIZECATEGORICAL Normalize categorical values for comparison.
0513         %
0514         %   Converts values to lowercase, trimmed strings (missing
0515         % become ""). For a sex-like variable, common short codes and
0516         % spellings are mapped to 'female', 'male', or 'other'.
0517         %
0518         % Inputs:
0519         %   vals   - String array, char, numeric, or cell array of values
0520         %   varKey - Variable name; a name containing 'sex' enables the sex
0521         %            mapping
0522         %
0523         % Outputs:
0524         %   out - String array the size of vals
0525         %
0526         % Toolbox Dependencies: None
0527         % normalize input values (vals can be string array, char, numeric,
0528         % cell)
0529         % varKey: name of the variable (e.g., 'sex','race') to allow
0530         % special-casing
0531         s = string(vals); % convert to string array (handles mixed types)
0532         s = lower(strtrim(s));         % lowercase and trim whitespace
0533         % replace missing with empty string
0534         s(ismissing(s)) = "";
0535         out = s;
0536         % special mapping for sex-like fields
0537         if contains(lower(varKey),'sex')
0538             femaleTokens = ["f","female","fem","woman","w"];
0539             maleTokens   = ["m","male","man"];
0540             otherTokens  = ["other","o","nonbinary","non-binary","nb", ...
0541                 "genderqueer","trans","nonbinary"];
0542             for ii = 1:numel(out)
0543                 v = out(ii);
0544                 if v=="" , continue; end
0545                 % strip punctuation/spaces
0546                 cleaned = regexprep(char(v),'[^a-z0-9]','');
0547                 cleaned = string(cleaned);
0548                 if any(cleaned == femaleTokens)
0549                     out(ii) = "female";
0550                 elseif any(cleaned == maleTokens)
0551                     out(ii) = "male";
0552                 elseif any(cleaned == otherTokens)
0553                     out(ii) = "other";
0554                 else
0555                     out(ii) = cleaned; % fallback lowercase cleaned token
0556                 end
0557             end
0558         else
0559             % generic cleaning for other categorical values
0560             for ii = 1:numel(out)
0561                 v = out(ii);
0562                 if v=="" , continue; end
0563                 cleaned = regexprep(char(v),'^\s+|\s+$','');    % trim ends
0564                 % collapse interior whitespace
0565                 cleaned = regexprep(cleaned,'\s+',' ');
0566                 out(ii) = lower(string(cleaned));
0567             end
0568         end
0569     end

0571 %% Unit conversion helper
0572 % - Note: convertUnits(varName, fromUnit, toUnit, vals)
0573 % - Supports height units: meters, centimeters, inches
0574 % - Supports weight units: kilograms, pounds
0575     function out = convertUnits(varName, fromUnit, toUnit, vals)
0576         %CONVERTUNITS Convert height or weight values between units.
0577         %
0578         %   Height units: meters, centimeters, inches, feet, and 'Feet and
0579         % Inches' (N-by-2 [feet, inches]). Weight units: kilograms,
0580         % pounds. An unrecognized input unit is taken as meters (height)
0581         % or kilograms (weight); a variable that is neither height nor
0582         % weight, or an empty or identical unit pair, returns vals
0583         % unchanged.
0584         %
0585         % Inputs:
0586         %   varName  - Variable name containing 'height' or 'weight'
0587         %   fromUnit - Unit of vals
0588         %   toUnit   - Unit to convert to
0589         %   vals     - Numeric values to convert
0590         %
0591         % Outputs:
0592         %   out - Converted values; N-by-2 [feet, inches] for
0593         %         'Feet and Inches'
0594         %
0595         % Toolbox Dependencies: None
0596         out = vals;
0597         if isempty(fromUnit) || isempty(toUnit) || ...
0598                 strcmpi(fromUnit,toUnit), return; end
0599         lname = lower(varName);
0600         if contains(lname,'height')
0601             % normalize: convert 'fromUnit' values into meters, then to
0602             % 'toUnit'
0603             switch lower(fromUnit)
0604                 case 'meters'
0605                     meters = vals;
0606                 case 'centimeters'
0607                     meters = vals/cmPerMeter;
0608                 case 'inches'
0609                     meters = vals*metersPerInch;
0610                 case {'feet', 'ft'}
0611                     meters = vals*metersPerFoot;
0612                 case {'feet and inches', 'ftin', 'ft/in'}
0613                     % Expect vals as Nx2: [feet, ...
0614                     %    inches] or a single value (total inches)
0615                     if ismatrix(vals) && size(vals,2) == 2
0616                         meters = vals(:,1)*metersPerFoot + ...
0617                             vals(:,2)*metersPerInch;
0618                     else
0619                         % fallback: treat as total inches
0620                         meters = vals*metersPerInch;
0621                     end
0622                 otherwise
0623                     meters = vals; % assume already meters if unknown
0624             end
0625             switch lower(toUnit)
0626                 case 'meters'
0627                     out = meters;
0628                 case 'centimeters'
0629                     out = meters*cmPerMeter;
0630                 case 'inches'
0631                     out = meters/metersPerInch; % meters -> inches
0632                 case {'feet', 'ft'}
0633                     out = meters/metersPerFoot;
0634                 case {'feet and inches', 'ftin', 'ft/in'}
0635                     % Return Nx2: [feet, inches]
0636                     feet = floor(meters/metersPerFoot);
0637                     inches = (meters - feet*metersPerFoot)/metersPerInch;
0638                     out = [feet, inches];
0639                 otherwise
0640                     out = meters;
0641             end
0642         elseif contains(lname,'weight')
0643             % convert 'fromUnit' values into kilograms, then to 'toUnit'
0644             switch lower(fromUnit)
0645                 case 'kilograms', kg = vals;
0646                 case 'pounds', kg = vals*kgPerPound;
0647                 otherwise, kg = vals; % assume kilograms if unknown
0648             end
0649             switch lower(toUnit)
0650                 case 'kilograms', out = kg;
0651                 case 'pounds', out = kg / kgPerPound;
0652                 otherwise, out = kg;
0653             end
0654         end
0655     end
